FILE check_cycle.py
1: """Standard-library certificate: integral worth, positive floor, target pinning, ideal."""
2: from fractions import Fraction as F
3: from itertools import product
4: from pathlib import Path
5: import json
6: 
7: def require(ok,msg):
8:     if not ok:raise RuntimeError(msg)
9: 
10: def check(r):
11:     m,k=r['m'],r['k'];n=m*k;T=F(r['T']);tau=F(r['floor']);target=r['target']
12:     def w(t):return sum(A[t[a]][t[b]]for (a,b),A in zip(r['edges'],r['matrices']))
13:     teams=list(product(range(m),repeat=k))
14:     matching=r['optimal_matching']
15:     require(len(matching)==m and all(sorted(t[v]for t in matching)==list(range(m))for v in range(k)),'Not perfect matching')
16:     require(sum(w(t)for t in matching)==T,'Matching worth')
17:     def covers(point):
18:         require(len(point)==n and min(point)>=0 and sum(point)==T,'Efficiency or nonnegativity')
19:         for t in teams:require(sum(point[v*m+t[v]]for v in range(k))>=w(t),'Uncovered team')
20:     x=list(map(F,r['x']));y=list(map(F,r['ideal_witness']));covers(x);covers(y)
21:     require(min(x)==tau>0,'Positive floor')
22:     loads=[F(0)]*n;dual_w=F(0)
23:     for row in r['residual_dual']:
24:         t=row['team'];weight=F(row['weight'])
25:         require(tuple(t)in teams and weight>=0,'Invalid dual term')
26:         for v in range(k):loads[v*m+t[v]]+=weight
27:         dual_w+=weight*w(t)
28:     d=[1-v for v in loads];require(min(d)>=0 and d[target]>0,'Dual loads or target deficit')
29:     require(T-dual_w==tau*sum(d),'Dual objective identity')
30:     # For every core p: sum(d_i p_i)+sum(lambda_t*(p(t)-w(t)))=T-lambda.w.
31:     # All terms are nonnegative. Thus p>=tau forces p_i=tau whenever d_i>0.
32:     target_team=next(t for t in matching if t[target//m]==target%m)
33:     H=w(target_team)
34:     require(y[target]==H and x[target]==tau,'Ideal attainment or target payoff')
35:     require(tau/H==F(r['ratio'])<F(1,k),'Protection ratio')
36:     return {'status':'PASS','team_cover_checks':2*len(teams),'target':target,'ratio':str(tau/H),'ideal':str(H),'dual_terms':len(r['residual_dual'])}
37:
