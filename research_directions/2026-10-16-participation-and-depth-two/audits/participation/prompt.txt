1: """Exact independent max-profit flow oracle for bipartite resource graphs."""
2: from fractions import Fraction as Q

4: def value(cap,edges):
5:     d=len(cap);neighbors=[[] for _ in range(d)]
6:     for u,v,p in edges:
7:         u,v=int(u),int(v);neighbors[u].append(v);neighbors[v].append(u)
8:     color={}
9:     for root in range(d):
10:         if root in color:continue
11:         color[root]=0;stack=[root]
12:         while stack:
13:             u=stack.pop()
14:             for v in neighbors[u]:
15:                 if v not in color:color[v]=1-color[u];stack.append(v)
16:                 elif color[v]==color[u]:raise RuntimeError('non-bipartite graph')
17:     N=d+2;s=d;t=d+1;graph=[[] for _ in range(N)]
18:     def add(u,v,c,k):
19:         graph[u].append([v,Q(c),Q(k),len(graph[v])]);graph[v].append([u,Q(0),-Q(k),len(graph[u])-1])
20:     for u,c in enumerate(cap):
21:         if color[u]==0:add(s,u,c,0)
22:         else:add(u,t,c,0)
23:     total=sum(cap)
24:     for u,v,p in edges:
25:         u,v=int(u),int(v)
26:         if color[u]==1:u,v=v,u
27:         add(u,v,total,-Q(p))
28:     objective=Q(0)
29:     while True:
30:         dist=[None]*N;dist[s]=Q(0);prev=[None]*N
31:         for _ in range(N-1):
32:             change=False
33:             for u in range(N):
34:                 if dist[u] is None:continue
35:                 for k,(v,c,cost,rev) in enumerate(graph[u]):
36:                     if c>0 and (dist[v] is None or dist[u]+cost<dist[v]):dist[v]=dist[u]+cost;prev[v]=(u,k);change=True
37:             if not change:break
38:         if dist[t] is None or dist[t]>=0:break

41:             u,k=prev[node];path.append((u,k));node=u

44:         for u,k in path:
45:             v,c,cost,rev=graph[u][k];graph[u][k][1]-=amount;graph[v][rev][1]+=amount
