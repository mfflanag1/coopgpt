# Python3, standard library only. Generates and checks the closed-form certificates.
from fractions import Fraction as Q
from pathlib import Path
import json

def family(L):
 assert L>=4 and L%2==0
 for h in range(2,L,2):
  for typ in 'ABCD':
   holes={'A':{h-1,h,h+1},'B':{1,h,L-1},'C':{0,h,h+1},'D':{0,h-1,h}}[typ]
   independent=[]
   for j in range(L):
    if j in holes:continue
    if typ=='A':top=(j%2==0) if j<h else (j%2==1)
    elif typ=='B':top=(j%2==1) if j<h else (j%2==0)
    elif typ=='C':top=j%2==1
    else:top=j%2==0
    independent.append(j if top else j+L)
   yield {'type':typ,'h':h,'independent_set':sorted(independent),'weight':str(Q(1,2*(L-2)))}

def check(L,save=False):
 n=2*L;k=L+3;star={0,1,L,n-1};support=list(family(L))
 ss=[set(x['independent_set']) for x in support]
 assert len(ss)==2*(L-2)
 for s in ss:
  assert len(s)==L-3 and len(s&star)==1
  assert all((i+1)%n not in s and (i+L)%n not in s for i in s)
 marg=[Q(sum(i in s for s in ss),len(ss)) for i in range(n)]
 assert all(marg[i]==(Q(1,4) if i in star else Q(L-4,2*L-4)) for i in range(n))
 w=[1-x for x in marg];t=sum(x*x for x in w)
 assert t==Q(9,4)+Q(L*L,2*(L-2))
 assert all(sum(w[i] for i in range(n) if i not in s)==t for s in ss)
 assert 3*Q(3,4)+(k-3)*Q(L,2*(L-2))==t
 rho=Q(n,k*k);H=rho*n*Q(9,16)
 if save:
  cert={'status':'exact_verified_closed_form','n':n,'L':L,'graph':f'vertices Z/{n}Z; cyclic neighbors and antipodes +{L}','row_universe':f'complements of ALL independent sets of cardinality{L-3}','focal':0,'closed_star':sorted(star),'support':support,'parameters':{'row_size':k,'rho':str(rho),'theta':'3/4','H':str(H),'t':str(t),'dual_mass':str(1/t),'focal_payment':str(Q(3,4)/t)}}
  path=Path(__file__).parent/f'activation_mass_simple_certificate_{n}.txt'
  path.write_text(json.dumps(cert,indent=2)+'\n')
 return H

if __name__=='__main__':
 for L in [4,6,8,10,48,50,52,86,100,500]:
  H=check(L,L in (50,52,86))
  print('PASS',{'L':L,'n':2*L,'supports':2*(L-2),'H':str(H)})
