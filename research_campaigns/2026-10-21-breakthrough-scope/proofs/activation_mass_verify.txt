# Run with Python3; requires only standard library. Reads both sibling certificates.
import json
from fractions import Fraction as Q
from pathlib import Path
for name in ('activation_mass_certificate.txt','activation_mass_certificate_172.txt'):
 c=json.loads((Path(__file__).parent/name).read_text())
 n=c['n'];L=c['L'];star={0,1,L,n-1};k=L+3
 assert n==2*L and L%2==0
 supp=[(frozenset(r['independent_set']),Q(r['weight'])) for r in c['support']]
 assert len({s for s,a in supp})==len(supp)
 assert all(a>0 and len(s)==L-3 and len(s&star)==1 for s,a in supp)
 assert all(all((i+1)%n not in s and (i+L)%n not in s for i in s) for s,a in supp)
 assert sum(a for s,a in supp)==1
 v=[sum(a for s,a in supp if i in s) for i in range(n)]
 assert all(v[i]==(Q(1,4) if i in star else Q(1,2)-Q(2,n-4)) for i in range(n))
 w=[1-x for x in v];t=sum(x*x for x in w)
 assert all(sum(w[i] for i in range(n) if i not in s)==t for s,a in supp)
 # All off-owner rows complement independent sets containing0, hence have3star vertices.
 assert 3*w[0]+(k-3)*w[2]==t
 rho=Q(n,k*k);H=rho*n*Q(9,16)
 assert H>2
 assert Q(c['parameters']['rho'])==rho
 assert Q(c['parameters']['H'])==H
 assert Q(c['parameters']['t'])==t
 assert Q(c['parameters']['dual_mass'])==1/t
 assert Q(c['parameters']['focal_payment'])==Q(3,4)/t
 print(name,'PASS',{'support':len(supp),'rho':str(rho),'H':str(H),'t':str(t),'minweight':str(min(a for s,a in supp))})
