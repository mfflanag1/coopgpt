\date{July 2026 --- working fragment}
\begin{document}
\maketitle
\noindent\emph{Status: mathematical core of a paper in progress. Singleton
bumps are proved monotone even on balanced games; grand-coalition bumps are
proved to remain on an exact path, with labeled resource monotonicity still
open. Proper nonsingleton bumps are reduced to an exact capacity dichotomy and
one open path lemma. Balanced nonexact counterexamples establish that
exactness is necessary.}
